"""
NAVIGATION AGENT
================
Deterministic interview navigation.

The agent proposes an action. The Orchestrator validates and executes it.
Interaction limits are tracked per item by the Orchestrator.

Policy:
- A participant who does not understand the question receives another
  explanation/rephrasing; this intent alone never causes an automatic skip.
- Off-topic or unclear replies get one redirect; a second unsuccessful
  attempt skips the item.
- Cannot-answer replies get one gentle follow-up; a second one skips,
  unless the guide says to move on immediately.
- A refusal gets one non-pressuring offer to help or skip; a second refusal
  skips. An explicit skip command should still be honoured immediately
  by the Orchestrator.
- Sufficient answers advance normally.
"""
from typing import Literal

from pydantic import BaseModel

NavigationAction = Literal["advance", "branch", "clarify", "rephrase", "skip"]
SkipReason = Literal["refused", "cannot_answer", "unresolved"]
ClarifyMode = Literal["clarify", "gentle", "redirect", "rephrase"]


class NavigationDecision(BaseModel):
    """A proposed next action plus the rule that produced it."""

    action: NavigationAction
    reason: str
    next_step: str | None = None
    skip_reason: SkipReason | None = None
    question_mode: ClarifyMode | None = None


def decide_next_action(
    intent: str,
    sufficiency: str,
    clarification_count: int,
    max_clarifications: int,
    next_step: str | None = None,
    cannot_answer_moves_on: bool = False,
) -> NavigationDecision:
    """Propose the next action; clarification is not limited by a numeric budget."""

    valid_intents = {
        "answer",
        "clarification_request",
        "cannot_answer",
        "refusal",
        "off_topic",
        "unclear",
    }
    valid_sufficiency = {"sufficient", "ambiguous", "insufficient"}

    if intent not in valid_intents:
        raise ValueError(f"Unknown intent: {intent!r}")
    if sufficiency not in valid_sufficiency:
        raise ValueError(f"Unknown sufficiency: {sufficiency!r}")

    # EMSAG: Keep the item-specific SCID rule for cannot-answer responses.
    if intent == "cannot_answer" and cannot_answer_moves_on:
        return NavigationDecision(
            action="skip",
            skip_reason="cannot_answer",
            reason="Source rule for this item: if the participant cannot answer, move on.",
        )

    if intent == "answer" and sufficiency == "sufficient":
        return NavigationDecision(
            action="advance",
            next_step=next_step,
            reason="Enough information to assess the item.",
        )

    # EMSAG: Allow repeated clarification and rephrasing without a numeric limit.
    if intent == "clarification_request":
        return NavigationDecision(
            action="rephrase",
            question_mode="rephrase",
            reason="Participant requests clarification; rephrasing is allowed.",
        )

    if intent == "refusal":
        return NavigationDecision(
            action="skip",
            skip_reason="refused",
            reason="Participant declined to answer.",
        )

    if intent == "cannot_answer":
        return NavigationDecision(
            action="clarify",
            question_mode="gentle",
            reason="Offer a gentle follow-up when the participant cannot answer.",
        )

    if intent in ("off_topic", "unclear"):
        return NavigationDecision(
            action="clarify",
            question_mode="redirect",
            reason=f"Redirect the participant to the current item after a {intent} response.",
        )

    return NavigationDecision(
        action="clarify",
        question_mode="clarify",
        reason=f"Answer is {sufficiency}; ask about the missing information.",
    )

"""
INTERVIEW ORCHESTRATOR
========================
Coordinates one participant turn. It is the ONLY component that changes the
interview state (InterviewGuide) — the agents only return proposals/text:

  participant text
     │
     ├─ 0. paused?  → fixed "paused" message (after repeated technical failures)
     ├─ 1. deterministic explicit-skip check (skip_phrases) → refusal, no LLM needed
     ├─ 2. Judgment Agent  → intent + sufficiency (+ provisional rating, evidence)
     │        └─ failure → technical-error path (no advance, no budget used)
     ├─ 3. Navigation Agent → PROPOSED action
     ├─ 4. validate proposal against the guide (budget, transition target)
     ├─ 5. execute:
     │        clarify/rephrase → Question Agent text (validated, fallback) and
     │                           one unit of the clarification budget
     │        advance          → item record "answered", short acknowledgement,
     │                           configured successor
     │        skip             → item record "skipped" + reason, fixed neutral
     │                           message, configured successor
     └─ 6. save the session record

Safety: the keyword safety check runs in app.py BEFORE this function on every
message and is independent of everything here; refusal/clarification handling
can therefore never disable it.

Model/validation failures are kept apart from participant behaviour: they
are logged as system events, the participant is asked to resend, the
clarification budget is not used, and after MAX_CONSECUTIVE_SYSTEM_ERRORS the
interview pauses instead of skipping content.
"""
import re
from dataclasses import dataclass, field

from agents.judgment_agent import JudgmentError, judge_answer
from agents.navigation_agent import NavigationDecision, decide_next_action
from agents.question_agent import generate_clarification_question
from session_record import build_item_record

MAX_CONSECUTIVE_SYSTEM_ERRORS = 3

LANGUAGE_NAMES = {"de": "German", "en": "English"}


@dataclass
class TurnOutcome:
    """What the application should show after a turn.

    feedback:   acknowledgement / skip notice / error message (may be "")
    next_text:  the next prompt (item, clarification or closing), or None
    next_kind:  "question" | "closing" | None (used for "repeat question")
    trace:      decision details for the console log
    """
    feedback: str
    next_text: str | None
    next_kind: str | None
    trace: dict = field(default_factory=dict)


def is_explicit_skip(answer_text: str, skip_phrases: list[str]) -> bool:
    """
    Deterministic detection of an explicit skip command such as
    "überspringen" or "Bitte überspringen." (exact phrase, optionally with
    "bitte"/"please"). Kept strict on purpose so that ordinary answers
    containing these words are not mistaken for a skip; anything less
    explicit is left to the Judgment Agent (intent "refusal").
    """
    text = re.sub(r"[^\w\s]", " ", answer_text.lower())
    text = re.sub(r"\b(bitte|please|einfach|ich möchte|i want to|let's|lass uns)\b", " ", text)
    text = re.sub(r"\s+", " ", text).strip()
    return text in skip_phrases



def validate_decision(decision, guide):
    # EMSAG: RT01: Allow unlimited clarification, but respect explicit item-level restrictions.
    if decision.action in ("clarify", "rephrase"):
        question = guide.current_question()
        if question is not None and not question.get("allow_clarification", True):
            return NavigationDecision(
                action="skip",
                skip_reason="unresolved",
                reason="Clarification is disabled for this SCID item.",
            )

    if decision.action == "advance":
        return decision.model_copy(
            update={"next_step": guide.next_item_id()}
        )

    if decision.action == "branch":
        if (
            decision.next_step is None
            or not guide.has_item(decision.next_step)
        ):
            raise ValueError(
                f"Invalid branch target: {decision.next_step}"
            )

    return decision


def handle_turn(llm, guide, recorder, answer_text: str, acknowledge=None) -> TurnOutcome:
    """
    Process one participant message for the current item.

    Args:
        llm: The model used by the Judgment and Question Agents.
        guide: InterviewGuide (state; changed only here).
        recorder: SessionRecorder (researcher-only JSON record).
        answer_text: The participant's message.
        acknowledge: Optional callable (prompt_text, answer_text) -> str that
            produces a short acknowledgement after a sufficient answer (the
            existing behavior-profile feedback chain). Falls back to a fixed text.

    Returns:
        TurnOutcome for the application to display.
    """
    item = guide.current_question()
    if item is None:
        return TurnOutcome("", guide.closing_message, "closing", {"action": "finished"})
    if guide.paused:
        return TurnOutcome(guide.messages["paused"], None, None, {"action": "paused"})

    #EMSAG: RT07: what if user wants to end the conversation?
    # Explicit request to end the entire interview.
    normalized_answer = " ".join(
        re.sub(r"[^\w\s]", " ", answer_text.casefold()).split()
    )

    stop_phrases = {
        "de": {
            "ich möchte das gespräch beenden",
            "ich will das gespräch beenden",
            "ich möchte aufhören",
            "ich möchte das interview beenden",
            "beende das gespräch",
        },
        "en": {
            "i want to end the conversation",
            "i want to end the interview",
            "i would like to stop",
            "please end the conversation",
        },
    }

    if normalized_answer in stop_phrases.get(guide.language, set()):
        guide.record_participant_turn(answer_text)
        guide._move_to(None)
        recorder.save()

        return TurnOutcome(
            "Natürlich. Wir beenden das Gespräch hier. Vielen Dank für Ihre Zeit."
            if guide.language == "de"
            else "Of course. We will end the conversation here. Thank you for your time.",
            "",
            "closing",
            {"action": "end_interview"},
        )
    #EMSAG: RT07: what if user wants to end the conversation?
    language = LANGUAGE_NAMES.get(guide.language, "German")
    guide.record_participant_turn(answer_text)

    # 1. Deterministic skip command: honoured even if the model is down.
    judgment = None
    if is_explicit_skip(answer_text, guide.skip_phrases):
        decision = NavigationDecision(action="skip", skip_reason="refused",
                                      reason="Explicit skip command (deterministic check).")
    else:
        # 2. Judgment.
        try:
            judgment = judge_answer(llm, item, guide.item_turns)
        except JudgmentError as exc:
            return _technical_failure(guide, recorder, item, str(exc))
        # 3. Navigation proposal.
        decision = decide_next_action(
            intent=judgment.intent,
            sufficiency=judgment.sufficiency,
            clarification_count=guide.current_clarification_count(),
            max_clarifications=guide.max_clarifications(),
            next_step=item.get("next_step"),
            cannot_answer_moves_on=item.get("cannot_answer_moves_on", False),
        )
    guide.consecutive_system_errors = 0

    # 4. Validation of the proposal.
    proposed = decision
    decision = validate_decision(decision, guide)
    trace = {
        "item": item["id"],
        "judgment": judgment.model_dump() if judgment else None,
        "proposed": proposed.model_dump(),
        "executed": decision.model_dump(),
        "clarifications_used": guide.current_clarification_count(),
    }

    # 5. Execution.
    if decision.action in ("clarify", "rephrase"):
        text = generate_clarification_question(
            llm, item, guide.item_turns,
            judgment.missing_information if judgment else [],
            mode=decision.question_mode or "clarify", language=language,
        )
        # EMSAG: Changed fallback handling to prevent valid Question Agent responses
        # from being overwritten by hardcoded text.
        if text is None:
            # Deterministic fallback when the Question Agent fails twice.
            recorder.log_system_event(
                item["id"],
                "question_agent_fallback",
                "invalid or failed output",
            )
            #EMSAG: RT02: "anders gefragt:" ist not used anymore
            if decision.question_mode == "rephrase":
                text = (
                    "Ich meine, ob Sie die Frage in Ihren eigenen Worten "
                    "beschreiben können. Was bedeutet das für Sie?"
                    if guide.language == "de"
                    else "I mean, could you describe the question in your own words? "
                        "What does it mean to you?"
                )
            elif decision.question_mode == "redirect":
                text = (
                    "Lassen Sie uns zu meiner Frage zurückkommen: "
                    f"{item['text']}"
                    if guide.language == "de"
                    else "Let us return to my question: "
                        f"{item['text']}"
                )
            elif decision.question_mode == "gentle":
                text = (
                    "Sie können sich gern etwas Zeit nehmen. "
                    "Was fällt Ihnen dazu ein?"
                    if guide.language == "de"
                    else "Take your time. What comes to mind?"
                )
            else:
                text = (
                    item["text"]
                )
            #EMSAG: RT02: "anders gefragt:" ist not used anymore
            trace["question_fallback"] = True
        else:
            trace["question_fallback"] = False

        print(
            f"[orchestrator] mode={decision.question_mode}, "
            f"fallback={trace['question_fallback']}, response={text!r}"
        )

        # EMSAG: Changed fallback handling to prevent valid Question Agent responses
        # from being overwritten by hardcoded text.
        guide.start_clarification(text, kind=decision.question_mode or decision.action)
        recorder.save()
        return TurnOutcome("", text, "question", trace)

    clarifications_used = guide.current_clarification_count()
    if decision.action in ("advance", "branch"):
        record = build_item_record(item, list(guide.item_turns), "answered", judgment=judgment,
                                   decision=decision, clarifications_used=clarifications_used)
        feedback = _acknowledgement(acknowledge, guide, answer_text)
        next_item = guide.finish_item(record, decision.next_step)
    #EMSAG: RT04: Skip Refusal Conditions were improved
    else:  # skip
        record = build_item_record(
            item,
            list(guide.item_turns),
            "skipped",
            judgment=judgment,
            skip_reason=decision.skip_reason,
            decision=decision,
            clarifications_used=clarifications_used,
        )

        if decision.skip_reason == "refused":
            feedback = (
                "Das ist völlig in Ordnung. "
                "Wir können zur nächsten Frage übergehen."
                if guide.language == "de"
                else "That's completely fine. "
                    "We can move on to the next question."
            )
        else:
            feedback = _acknowledgement(
                acknowledge, guide, answer_text
            )

        next_item = guide.finish_item(record, guide.next_item_id())
    #EMSAG: RT04: Skip Refusal Conditions were improved

    recorder.save()
    if next_item is None:
        return TurnOutcome(feedback, guide.closing_message, "closing", trace)
    return TurnOutcome(feedback, guide.presentation_text(next_item), "question", trace)


def _default_prefix(guide) -> str:
    """Lead-in for the deterministic fallback reformulation (used only when the
    Question Agent fails twice): the original item is shown again with this prefix."""
    return "Anders gefragt:" if guide.language == "de" else "To put it differently:"

#EMSAG: RT04: not only saying danke but also give more empathy and acknowladgement
def _acknowledgement(acknowledge, guide, answer_text: str) -> str:
    """Return a meaningful acknowledgement without generic thanks alone."""
    if acknowledge is None:
        return guide.messages["acknowledge"]

    try:
        text = (acknowledge(guide.active_prompt_text(), answer_text) or "").strip()
    except Exception:
        text = ""

    # An acknowledgement must not contain a follow-up question.
    if not text or "?" in text:
        text = ""

    # Reject generic thanks as the entire acknowledgement.
    normalized = " ".join(text.casefold().split()).strip(" .!,:;")
    generic_thanks = {
        "danke",
        "vielen dank",
        "danke für ihre antwort",
        "danke für ihre offenheit",
        "thank you",
        "thanks",
        "thank you for your answer",
        "thanks for sharing",
    }

    if normalized in generic_thanks:
        text = ""

    if text:
        return text

    # Use a context-sensitive fallback for clearly emotional disclosures.
    answer_lower = answer_text.casefold()
    emotional_terms = (
        "einsam", "allein", "niemand", "traurig", "tod",
        "gestorben", "hoffnungslos", "wertlos", "misshandelt",
        "gewalt", "verletzt", "verzweifelt",
    )

    if any(term in answer_lower for term in emotional_terms):
        return "Es klingt belastend, was Sie beschreiben."

    # Preserve the existing fallback for other answers.
    return guide.messages["acknowledge"]
#EMSAG: RT04: not only saying danke but also give more empathy and acknowladgement


def _technical_failure(guide, recorder, item, detail: str) -> TurnOutcome:
    """
    Handle a failed/malformed model response: do not advance, do not spend
    budget, do not keep the unjudged message as evidence; ask to resend;
    pause after repeated failures.
    """
    guide.item_turns.pop()  # the message will be judged again when resent
    guide.consecutive_system_errors += 1
    recorder.log_system_event(item["id"], "judgment_failure", detail)
    if guide.consecutive_system_errors >= MAX_CONSECUTIVE_SYSTEM_ERRORS:
        guide.paused = True
        recorder.save()
        return TurnOutcome(guide.messages["paused"], None, None, {"action": "paused", "detail": detail})
    recorder.save()
    return TurnOutcome(guide.messages["system_error"], guide.active_prompt_text(), "question",
                       {"action": "system_error", "detail": detail})

"""
INTERVIEW GUIDE
=================
Reads one interview guide file from study_content/interviews/*.yaml and
keeps the explicit INTERVIEW STATE for the current participant: which item
is active, what the participant has said about it so far, how much of the
clarification budget is used, and the finished per-item records.

This file deliberately knows nothing about the LLM, audio, or the web
server. It is the state store that the Orchestrator
(backend/agents/interview_orchestrator.py) reads and changes. The
Navigation Agent only PROPOSES actions; the Orchestrator validates them and
then calls the state-changing methods here (begin_clarification(),
finish_item(), ...). Nothing in this file decides what an answer means.

Per the project's core principle "the LLM presents, it does not invent":
question wording always comes from the YAML file. The Question Agent may
explain or reformulate the CURRENT item, but never adds new items.

Item fields (all optional except id/text — see the example guide's header):
  id, text, criterion_id     as before
  criterion                  human-readable description of what the item
                             assesses (passed to the Judgment/Question Agents)
  rating_scheme              "none" (default) or "scid5pd" (? / 0 / 1 / 2)
  rating_anchor              the source's threshold anchor, e.g. what a "2"
                             requires; only used with a rating scheme
  probes                     source follow-up probes for this item (context
                             for the Question Agent; not asked automatically)
  max_clarifications         per-item clarification budget; falls back to the
                             guide's default_max_clarifications (default 1).
                             0 means "do not clarify, move on".
  next_step                  id of the item that follows, or "END". Validated
                             at load time. Missing = the next item in order.

SCID-5-PD support (see study_content/interviews/scid5pd_de.yaml header):
  criterion_code             source code, e.g. "PD27" (shared by questions of one criterion)
  intro                      source introduction shown before the item
  ask: false                 item not asked by the chatbot (observation /
                             clinician-only / derived / same_as / gate);
                             recorded automatically when reached
  gate                       source-defined decision point:
                             {criterion_codes, min_rated_2, if_met, if_not_met}
  cannot_answer_moves_on     source rule "IF CAN'T ANSWER, MOVE ON"
  safety_relevant            item concerns suicidality/self-harm (flag in record)

Guide-level fields added for the agent workflow:
  default_max_clarifications  project rule, default 1
  skip_phrases                exact explicit skip commands (deterministic,
                              works even if the LLM fails)
  interaction_messages        fixed texts for skips/technical problems

Used by: backend/app.py, backend/agents/interview_orchestrator.py.
"""
import os

import yaml

INTERVIEWS_DIR = os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "study_content", "interviews"
)

# Value used by next_step to mark the end of the interview.
END_MARKER = "END"

# Rating schemes the application understands. "scid5pd" is the coding of

# 1 = subthreshold, 2 = threshold.
RATING_SCHEMES = {
    "none": [],
    "scid5pd": ["?", "0", "1", "2"],
}

# Fallback fixed texts (project-specific wording, not SCID). A guide can

DEFAULT_SKIP_PHRASES = {
    "de": [
        "überspringen",
        "frage überspringen",
        "überspringe diese frage",
        "nächste frage",
        "weiter",

class GuideConfigError(ValueError):
    """Raised when a guide file is incomplete or inconsistent (e.g. an
    unfilled template item or a next_step pointing to a missing item)."""


class InterviewGuide:
    """Tracks one participant's progress through one interview guide file."""

    def __init__(self, guide_filename: str):
        """
        Load a guide file, flatten its items into one ordered list and
        validate the configured transitions.

        Args:
            guide_filename: File name inside study_content/interviews/,
                e.g. "depression_scid_example.yaml".

        Raises:
            GuideConfigError: if an item has no question text (e.g. an
                unfilled SCID template) or a next_step target does not exist.
        """
        path = os.path.join(INTERVIEWS_DIR, guide_filename)
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}

        self.guide_id = data.get("guide_id", guide_filename)
        self.title = data.get("title", "")
        self.behavior_profile = data.get("behavior_profile")

        #emsag 18.10.2026 extra changes:
        # Optional metadata for future interview-guide formats.
        # Defaults preserve compatibility with existing YAML files.
        self.guide_version = data.get("guide_version", "1.0")
        self.language = data.get("language", "de")
        self.interview_type = data.get("interview_type", "structured")
        self.navigation_mode = data.get("navigation_mode", "sequential")
        #emsag 18.10.2026 extra changes:

        # Content provenance label (e.g. "team-written mock, NOT SCID").
        # Copied into every session record so that results from mock content
        # can never be mistaken for SCID results.
        self.content_source = data.get("content_source", "unspecified")

        self.framing_message = (data.get("framing_message") or "").strip()
        self.closing_message = (data.get("closing_message") or "").strip()

        # Project rule (not a SCID rule): one clarification/reformulation
        # attempt per item unless the guide says otherwise.
        self.default_max_clarifications = max(0, int(data.get("default_max_clarifications", 1)))

        lang = self.language if self.language in DEFAULT_MESSAGES else "de"
        self.messages = dict(DEFAULT_MESSAGES[lang])
        self.messages.update(data.get("interaction_messages") or {})
        phrases = data.get("skip_phrases") or DEFAULT_SKIP_PHRASES.get(lang, [])
        self.skip_phrases = [str(p).strip().lower() for p in phrases if str(p).strip()]

        self.questions = []
        for section in data.get("sections", []):
            section_title = section.get("section_title")
            for question in section.get("questions", []):
                text = (question.get("text") or "").strip()
                ask = bool(question.get("ask", True))
                if ask and not text:
                    raise GuideConfigError(
                        f"Item {question.get('id')!r} in {guide_filename} has no question text. "
                        "Template items must be filled from the licensed source before use."
                    )
                rating_scheme = question.get("rating_scheme", data.get("rating_scheme", "none")) or "none"
                if rating_scheme not in RATING_SCHEMES:
                    raise GuideConfigError(
                        f"Item {question.get('id')!r}: unknown rating_scheme {rating_scheme!r}."
                    )
                self.questions.append({
                    "id": question["id"],
                    "text": text,
                    "section_title": section_title,
                    # Legacy single-follow-up switch (kept for compatibility;
                    # the agent workflow uses the clarification budget below).
                    "allow_followup": bool(question.get("allow_followup", False)),

                    # EMSAG 18.10.2026 Optional fields for future SCID-compatible guides.
                    "criterion_id": question.get("criterion_id", question["id"]),
                    # Clarification is allowed by default; the budget decides
                    # how often (max_clarifications: 0 disables it).
                    "allow_clarification": bool(question.get("allow_clarification", True)),
                    "max_clarifications": max(0, int(question.get(
                        "max_clarifications", self.default_max_clarifications))),
                    "next_step": question.get("next_step"),
                    # EMSAG 18.10.2026 Optional fields for future SCID-compatible guides.

                    "criterion": (question.get("criterion") or "").strip(),
                    "rating_scheme": rating_scheme,
                    "rating_anchor": (question.get("rating_anchor") or "").strip(),
                    "probes": [str(p).strip() for p in (question.get("probes") or []) if str(p).strip()],

                    # --- SCID-5-PD support -------------------------------------
                    # Source criterion code (e.g. "PD27"); several questions can
                    # share one code when the source asks several questions
                    # for one criterion.
                    "criterion_code": question.get("criterion_code"),
                    # Source introduction sentence shown once before the item.
                    "intro": (question.get("intro") or "").strip(),
                    # ask: false = item is NOT asked by the chatbot (observation
                    # items, clinician-only exclusion criteria, derived ratings,
                    # source decision points). It is processed automatically
                    # when reached, see _process_non_interactive().
                    "ask": ask,
                    "not_asked_reason": question.get("not_asked_reason"),
                    "note": (question.get("note") or "").strip(),
                    "same_as": question.get("same_as"),
                    "derived_any_rated_2_of": list(question.get("derived_any_rated_2_of") or []),
                    "gate": question.get("gate"),
                    # Source rule "IF CAN'T ANSWER, MOVE ON" for this item.
                    "cannot_answer_moves_on": bool(question.get("cannot_answer_moves_on", False)),
                    # Item touches suicidality/self-harm: flagged in the record
                    # for the clinician (the keyword safety check runs anyway).
                    "safety_relevant": bool(question.get("safety_relevant", False)),
                    # General guidance for ratings (e.g. SCID-5-PD p. 5-6),
                    # given to the Judgment Agent for rated items.
                    "rating_guidance": (data.get("rating_guidance") or "").strip(),
                })

        # Transitions are validated once at load time so that a typo in
        # next_step fails loudly here instead of silently ending or
        # misrouting an interview later.
        self._index_by_id = {q["id"]: i for i, q in enumerate(self.questions)}
        if len(self._index_by_id) != len(self.questions):
            raise GuideConfigError(f"Duplicate item ids in {guide_filename}.")
        for q in self.questions:
            target = q["next_step"]
            if target not in (None, END_MARKER) and target not in self._index_by_id:
                raise GuideConfigError(
                    f"Item {q['id']!r}: next_step {target!r} does not exist in {guide_filename}."
                )
            gate = q["gate"]
            if gate:
                for key in ("if_met", "if_not_met"):
                    if gate.get(key) not in (END_MARKER,) and gate.get(key) not in self._index_by_id:
                        raise GuideConfigError(
                            f"Gate {q['id']!r}: {key} target {gate.get(key)!r} does not exist.")
                if not gate.get("criterion_codes") or int(gate.get("min_rated_2", 0)) < 1:
                    raise GuideConfigError(f"Gate {q['id']!r} needs criterion_codes and min_rated_2.")
            if q["same_as"] and q["same_as"] not in self._index_by_id:
                raise GuideConfigError(f"Item {q['id']!r}: same_as {q['same_as']!r} does not exist.")

        self.reset()

    # ------------------------------------------------------------------
    # Current position
    # ------------------------------------------------------------------
    def current_question(self) -> dict | None:
        """Return the item that should be asked right now, or None if the guide is finished."""
        if self._current_index >= len(self.questions):
            return None
        return self.questions[self._current_index]

    def is_finished(self) -> bool:
        """True once the interview has reached its end (all items or next_step END)."""
        return self._current_index >= len(self.questions)

    def next_item_id(self) -> str | None:
        """
        The configured successor of the current item: its next_step if set,
        otherwise the next item in file order. Returns None for the end.
        Used by the Orchestrator to validate a proposed advance/branch.
        """
        question = self.current_question()
        if question is None:
            return None
        target = question.get("next_step")
        if target == END_MARKER:
            return None
        if target:
            return target
        nxt = self._current_index + 1
        return self.questions[nxt]["id"] if nxt < len(self.questions) else None

    def has_item(self, item_id: str) -> bool:
        """True if item_id exists in this guide (transition validation)."""
        return item_id in self._index_by_id

    def advance(self) -> dict | None:
        """
        Move to the configured next item (next_step, else file order) and
        reset all per-item state.

        Returns:
            The new current item, or None if the guide is now finished.
        """
        return self._move_to(self.next_item_id())

    def _move_to(self, item_id: str | None) -> dict | None:
        """
        Jump to item_id (None = end of interview), reset per-item state and
        automatically process any items the chatbot does not ask (see
        _process_non_interactive()). Returns the next item to ASK, or None.
        """
        self._current_index = self._index_by_id[item_id] if item_id else len(self.questions)
        self._reset_item_state()
        self._skip_non_interactive()
        return self.current_question()

    def _skip_non_interactive(self) -> None:
        """Record and pass over every consecutive ask:false item / source gate."""
        guard = 0
        while self.current_question() is not None and not self.current_question()["ask"]:
            item = self.current_question()
            record, target = self._process_non_interactive(item)
            self.records.append(record)
            self._current_index = self._index_by_id[target] if target else len(self.questions)
            self._reset_item_state()
            guard += 1
            if guard > len(self.questions):  # defensive: a misconfigured loop of gates
                raise GuideConfigError("Non-interactive items form a loop.")

    def _successor_of(self, item: dict) -> str | None:
        """Configured successor of an item (next_step or file order; None = end)."""
        target = item.get("next_step")
        if target == END_MARKER:
            return None
        if target:
            return target
        nxt = self._index_by_id[item["id"]] + 1
        return self.questions[nxt]["id"] if nxt < len(self.questions) else None

    def criterion_has_rating_2(self, criterion_code: str) -> bool:
        """
        True if any answered question recorded for criterion_code has a
        provisional rating of "2". Several SCID questions can belong to one
        criterion; the criterion is treated as "2" if one of its questions
        established the threshold ("?" and skipped items never count).
        """
        for rec in self.records:
            if rec.get("criterion_code") == criterion_code and rec.get("status") == "answered":
                if (rec.get("provisional_rating") or {}).get("value") == "2":
                    return True
        return False

    def _process_non_interactive(self, item: dict) -> tuple[dict, str | None]:
        """
        Build the record for an item the chatbot does not ask and return
        (record, next item id).

        * gate: SOURCE-DEFINED decision point (e.g. SCID-5-PD Antisocial PD
          "at least two Conduct Disorder criteria rated 2"). Evaluated on the
          PROVISIONAL chatbot ratings; the counted criteria are stored in the
          record so the clinician can verify the routing.
        * derived: rating derived from other criteria as stated in the source.
        * same_as / observation / clinician: recorded as
          "not assessed by chatbot – clinician rating required".
        """
        record = {
            "item_id": item["id"],
            "criterion_id": item.get("criterion_id"),
            "criterion_code": item.get("criterion_code"),
            "criterion": item.get("criterion"),
            "status": "not_asked",
            "not_asked_reason": item.get("not_asked_reason"),
            "note": item.get("note"),
        }
        gate = item.get("gate")
        if gate:
            codes = gate["criterion_codes"]
            rated_2 = [c for c in codes if self.criterion_has_rating_2(c)]
            met = len(rated_2) >= int(gate["min_rated_2"])
            target = gate["if_met"] if met else gate["if_not_met"]
            record.update({
                "status": "gate",
                "gate_met": met,
                "criteria_rated_2": rated_2,
                "min_rated_2": int(gate["min_rated_2"]),
                "routed_to": target,
                "note": ("Source-defined routing evaluated on PROVISIONAL chatbot ratings "
                         "('?' counts as not '2'); clinician to verify."),
            })
            return record, (None if target == END_MARKER else target)

        if item.get("derived_any_rated_2_of"):
            sources = item["derived_any_rated_2_of"]
            hit = [c for c in sources if self.criterion_has_rating_2(c)]
            record["status"] = "derived"
            record["provisional_rating"] = {
                "value": "2" if hit else "?", "scheme": "scid5pd",
                "label": "provisional – for clinician review",
                "derived_from": sources, "rated_2": hit,
            }
        elif item.get("rating_scheme", "none") != "none":
            record["provisional_rating"] = {"value": "?", "scheme": item["rating_scheme"],
                                            "label": "not assessed by chatbot – clinician rating required"}
        if item.get("same_as"):
            record["same_as"] = item["same_as"]
        return record, self._successor_of(item)

    def presentation_text(self, item: dict | None = None) -> str:
        """Text shown when an item is first presented: optional source intro + question."""
        item = item or self.current_question()
        if item is None:
            return ""
        return f"{item['intro']} {item['text']}".strip() if item.get("intro") else item["text"]

    def _reset_item_state(self) -> None:
        """Clear everything that belongs to the current item only."""
        self._awaiting_followup_answer = False
        self._followup_used_for_current = False
        self.pending_followup_text = None
        #EMSAG new parameters were added:
        self._clarifications_used_for_current = 0
        self._pending_followup_kind = None
        #EMSAG new parameters were added:
        # What the participant has said about the CURRENT item, in order,
        # paired with the prompt they were answering. Given to the Judgment
        # Agent so a clarification answer is judged together with the first
        # answer, not in isolation.
        self.item_turns: list[dict] = []
        self.consecutive_system_errors = 0

    # ------------------------------------------------------------------
    # Legacy single follow-up (kept for compatibility with older code/tests)
    # ------------------------------------------------------------------
    def start_followup(self, followup_text: str) -> None:
        """Record that a (legacy) follow-up question was asked for the current item."""
        self._awaiting_followup_answer = True
        self._followup_used_for_current = True
        self.pending_followup_text = followup_text
        self._pending_followup_kind = "followup"

    def is_awaiting_followup_answer(self) -> bool:
        """True if the participant's next answer responds to a follow-up/clarification."""
        return self._awaiting_followup_answer

    def followup_already_used(self) -> bool:
        """True if the legacy follow-up has already been used for the current item."""
        return self._followup_used_for_current

    # ------------------------------------------------------------------
    # Clarification budget
    # ------------------------------------------------------------------
    def max_clarifications(self) -> int:
        """Configured clarification budget of the current item (0 if clarification is disabled)."""
        question = self.current_question()
        if question is None or not question.get("allow_clarification", True):
            return 0
        return question.get("max_clarifications", self.default_max_clarifications)

    def current_clarification_count(self) -> int:
        """Number of clarifications/reformulations already used for the current item."""
        return self._clarifications_used_for_current

    def clarification_limit_reached(self) -> bool:
        """True if no further clarification is permitted for the current item."""
        return self._clarifications_used_for_current >= self.max_clarifications()

    #EMSAG: RT01: Clarification Request should not cut the budget
    # EMSAG: RT01: Allow unlimited clarification and rephrasing.
    def start_clarification(self, clarification_text: str, kind: str = "clarification") -> bool:
        """
        Store a clarification or rephrasing for the current item.
        Clarification count is tracked for documentation, not as a limit.
        """
        question = self.current_question()

        if question is None:
            return False

        # EMSAG: Respect explicit item-level clarification restrictions.
        if not question.get("allow_clarification", True):
            return False

        # EMSAG: Count every clarification, including rephrasing.
        self._clarifications_used_for_current += 1

        self._awaiting_followup_answer = True
        self._pending_followup_kind = kind
        self.pending_followup_text = clarification_text
        return True

    #EMSAG: RT01: Clarification Request should not cut the budget
    def pending_kind(self) -> str | None:
        """Kind of the pending clarification (public accessor; replaces reading _pending_followup_kind)."""
        return self._pending_followup_kind

    def clear_followup(self) -> None:
        """Clear the pending follow-up/clarification without advancing."""
        self._awaiting_followup_answer = False
        self.pending_followup_text = None
        self._pending_followup_kind = None

    # ------------------------------------------------------------------
    # Per-item evidence and records
    # ------------------------------------------------------------------
    def active_prompt_text(self) -> str:
        """The text the participant is currently answering (clarification if pending, else the item)."""
        question = self.current_question()
        if question is None:
            return ""
        return self.pending_followup_text or self.presentation_text(question)

    def record_participant_turn(self, answer_text: str) -> None:
        """Store one participant message for the current item, with the prompt it answered."""
        self.item_turns.append({"prompt": self.active_prompt_text(), "answer": answer_text})

    def finish_item(self, record: dict, next_item_id: str | None) -> dict | None:
        """
        Close the current item with a record and move to next_item_id.

        The Orchestrator must have validated next_item_id already (it must
        be None for the end, or an existing item id).

        Args:
            record: The item outcome (see session_record.build_item_record()).
            next_item_id: The validated successor.

        Returns:
            The new current item or None at the end.
        """
        self.records.append(record)
        return self._move_to(next_item_id)

    def reset(self) -> None:
        """Start the guide over from the first item (e.g. for a new participant)."""
        self._current_index = 0
        self.records: list[dict] = []
        self.paused = False
        self._reset_item_state()
        self._skip_non_interactive()

def _split_sentences(text: str) -> list[str]:
    """
    Split a piece of text into individual sentences, so each one can be
    streamed to the browser (and spoken) as soon as it's ready, instead of
    waiting for the whole piece. This matters because "the feedback" or "a
    question" are not guaranteed to always be exactly one sentence — if a
    prompt in study_content/prompts/ is edited later to allow a longer,
    multi-sentence feedback, this still splits it up correctly instead of
    silently becoming one large, slower chunk.

    This is a simple, punctuation-based split (not aware of abbreviations
    like "Dr." or "z.B.") — good enough for short interview responses. Piper
    does its own, more careful sentence splitting internally for the actual
    audio; this split only decides how text is grouped for streaming.
    """
    text = text.strip()
    if not text:
        return []
    parts = re.split(r'(?<=[.!?])\s+', text)
    return [p.strip() for p in parts if p.strip()]


def _tag_sentences(text: str, kind: str) -> list[dict]:
    """
    Like _split_sentences(), but each sentence is paired with a "kind"
    label ("feedback", "question", or "closing"). This is what lets the
    browser's "Frage wiederholen" (repeat question) feature replay only
    the actual question, not the feedback that came before it — see
    frontend/conversation.js's replayLastQuestion() and readStreamedReply().
    """
    return [{"text": s, "kind": kind} for s in _split_sentences(text)]


def _acknowledge(prompt_text: str, answer_text: str) -> str:
    """
    Short acknowledgement after a SUFFICIENT answer, using the existing
    behavior-profile feedback chain (study_content/prompts/). It is only
    called once the Orchestrator has decided to advance, so the wording can
    no longer contradict what happens next (previously it was generated
    before the answer was judged). Follow-ups are never requested here.
    """
    feedback, _ = llm_provider.get_feedback_and_followup(
        chain_with_history, profile, prompt_text, answer_text,
        followup_allowed=False, language_code=args.language,
    )
    return feedback


def process_answer(answer_text: str) -> list[dict]:
    """
    Process one participant message and return the reply as tagged sentences.

    All interview decisions are made by the Orchestrator
    (agents/interview_orchestrator.py): Judgment -> Navigation proposal ->
    validation -> execution, plus the researcher-only session record. This
    function only converts the outcome into the sentence pieces the text
    and voice endpoints already expect.
    """
    outcome = handle_turn(llm, interview, recorder, answer_text, acknowledge=_acknowledge)
    print(f"[agents] trace={json.dumps(outcome.trace, ensure_ascii=False, default=str)}")

    pieces = _tag_sentences(outcome.feedback, "feedback")
    if outcome.next_text:
        if (outcome.next_kind == "question" and outcome.feedback
                and profile.question_delivery == "natural_transition"):
            # Flexible profile: blend acknowledgement and next question (meaning preserved).
            combined = llm_provider.get_natural_transition(
                llm, outcome.feedback, outcome.next_text, language_code=args.language)
            return _tag_sentences(combined, "question")
        pieces += _tag_sentences(outcome.next_text, outcome.next_kind)
    return pieces


def _start_new_session() -> None:
    """Reset interview state, start a new session record and forget the
    previous participant's acknowledgement-chat history."""
    interview.reset()
    recorder.start()
    llm_provider.chat_sessions.clear()


@app.route("/api/interview/start", methods=["GET"])
def interview_start_endpoint():
    """
    Called once when the participant opens the page: returns the framing
    message and the first fixed question, verbatim from the active
    interview guide. No LLM call is involved — there is nothing to
    generate yet, only fixed text to present.
    Returns JSON: {"message": "..."}.
    """
    _start_new_session()
    first_question = interview.current_question()
    if first_question is None:
        # An empty guide (e.g. a guide file with no questions yet).

    return jsonify({"message": message})


@app.route("/api/interview/start/audio", methods=["GET"])
def interview_start_audio_endpoint():
    """
    Voice-condition only: pre-synthesizes the opening message (framing +
    first question) as audio, sentence by sentence. Called in the
    background while the start overlay is still showing (see
    frontend/conversation.js), so spoken playback can begin immediately
    once "Interview starten" is clicked, instead of only starting to
    synthesize at that point.

    Unlike /api/chat/audio, this is returned as one plain JSON object, not
    a stream — the whole point of fetching it in the background is that it
    is ready and waiting well before it's actually needed, so there's no
    time-to-first-sound pressure here the way there is for a live reply.

    Returns JSON: {"pieces": [{"text": "...", "audio": "...base64...", "kind": "..."}, ...]}.
    "kind" is "framing" or "question" — lets the browser's "Frage
    wiederholen" repeat just the question, not the greeting, consistent
    with how later turns are tagged (see process_answer()).
    """
    first_question = interview.current_question()

    tagged_sentences = _tag_sentences(interview.framing_message, "framing")
    if first_question is not None:
        tagged_sentences += _tag_sentences(interview.presentation_text(first_question), "question")

    pieces = []
    for sentence in tagged_sentences:
        sample_rate, audio_array = get_tts().synthesize(sentence["text"])
        audio_base64 = speech.encode_wav_base64(sample_rate, audio_array)
        pieces.append({"text": sentence["text"], "audio": audio_base64, "kind": sentence["kind"]})

    return jsonify({"pieces": pieces})


@app.route("/api/chat", methods=["POST"])
def chat_endpoint():
    """
    Text-condition endpoint: the participant typed an answer in the browser.
    Expects JSON: {"text": "..."}.
    Returns JSON: {"response": "...", "finished": bool} — see process_answer()
    for what "response" might contain (feedback + next question, feedback +
    a follow-up, or the closing message). "finished" is true once the
    entire guide has been completed (the closing message was just given,
    or already had been before this call) — the frontend uses this to stop
    offering further answers instead of silently repeating the closing
    message. "safety_flag" is true if the participant's own answer matched
    a configured safety keyword (see backend/safety_check.py) — the
    frontend shows a calm, non-blocking hint with support resources when
    true; the interview is not interrupted.
    """
    data = request.get_json(silent=True) or {}
    answer_text = (data.get("text") or "").strip()
    if not answer_text:
        return jsonify({"error": "Kein Text eingegeben."}), 400

    safety_flag = safety_check.check_for_safety_concern(answer_text, args.language)
    if safety_flag:
        print("[safety] Stichwort in der Antwort der teilnehmenden Person erkannt — Hinweis wird eingeblendet.")

    pieces = process_answer(answer_text)
    response = " ".join(p["text"] for p in pieces).strip()
    print(f"[interview] final response sent to browser: {response!r}")
    return jsonify({"response": response, "finished": interview.is_finished(), "safety_flag": safety_flag})


@app.route("/api/chat/audio", methods=["POST"])
def chat_audio_endpoint():
    """
    Voice-condition endpoint: the participant spoke into the browser's microphone.
    Expects multipart/form-data with an "audio" file and optional "language" field.

    Streams the reply back as NDJSON (one JSON object per line), instead of
    waiting for the whole reply to be synthesized before responding — this
    is what Step 5b of the project plan is for: the browser can start
    playing the first sentence while later ones are still being
    synthesized. Line types:

      {"type": "transcript", "transcript": "..."}
        Sent first, as soon as speech-to-text is done.
      {"type": "transcript", "transcript": "...", "safety_flag": bool}
        Sent first, as soon as speech-to-text is done. "safety_flag" is
        true if the participant's own words matched a configured safety
        keyword (see backend/safety_check.py) — the frontend shows a
        calm, non-blocking hint with support resources when true; the
        interview is not interrupted.
      {"type": "sentence", "text": "...", "audio": "...base64 wav...", "kind": "...", "is_last": bool, "finished": bool}
        One per sentence, in speaking order. "kind" is "feedback",
        "question", or "closing" — see process_answer() — and is what lets
        the browser's "Frage wiederholen" replay only the question later.
        "finished" is true once the entire guide has been completed —
        the frontend uses this to stop offering further answers instead
        of silently repeating the closing message.
      {"type": "error", "message": "..."}
        Sent instead of a "sentence" line if synthesizing one sentence
        fails; playback of anything already sent is unaffected.
    """
    audio_file = request.files.get("audio")
    if not audio_file:
        return jsonify({"error": "No audio file uploaded."}), 400

    language = request.form.get("language") or args.language
    try:
        audio_np = speech.decode_uploaded_wav(audio_file.read())
    except Exception as e:
        return jsonify({"error": f"Unable to decode audio: {e}"}), 400

    transcript = get_stt().transcribe(audio_np, language=language)
    safety_flag = safety_check.check_for_safety_concern(transcript, language)
    if safety_flag:
        print("[safety] Stichwort in der Antwort der teilnehmenden Person erkannt — Hinweis wird eingeblendet.")

    pieces = process_answer(transcript)
    finished = interview.is_finished()
    print(f"[interview] final response sent to browser: {' '.join(p['text'] for p in pieces)!r}")

    def generate():
        turn_start = time.time()

        def elapsed():
            return f"{time.time() - turn_start:.2f}s"

        print(f"[stream] transcript sent at +{elapsed()}")
        yield json.dumps({"type": "transcript", "transcript": transcript, "safety_flag": safety_flag}) + "\n"

        # Short pause appended after every sentence except the last, so
        # consecutive audio pieces played back-to-back in the browser still
        # sound like natural speech instead of running into each other —
        # same reasoning as the pause between sentences in backend/tts.py.
        gap_seconds = 0.3

        for i, piece in enumerate(pieces):
            is_last = (i == len(pieces) - 1)
            sentence_text = piece["text"]
            try:
                sample_rate, audio_array = get_tts().synthesize(sentence_text)
                if not is_last:
                    gap = np.zeros(int(sample_rate * gap_seconds), dtype=np.float32)
                    audio_array = np.concatenate([audio_array, gap])
                audio_base64 = speech.encode_wav_base64(sample_rate, audio_array)
                print(f"[stream] sentence {i + 1}/{len(pieces)} ({piece['kind']}) sent at +{elapsed()}: {sentence_text!r}")
                yield json.dumps({
                    "type": "sentence",
                    "text": sentence_text,
                    "audio": audio_base64,
                    "kind": piece["kind"],
                    "is_last": is_last,
                    "finished": finished,
                }) + "\n"
            except Exception as e:
                print(f"[stream] error at +{elapsed()}: {e}")
                yield json.dumps({"type": "error", "message": str(e)}) + "\n"
                break

    return Response(generate(), mimetype="application/x-ndjson")


# --- Terminal mode (used without --web) -------------------------------------
def run_terminal_mode():
    """
    A simple command-line loop: press Enter to record, speak, press Enter
    again to stop, and the assistant replies out loud. Useful for quick local
    testing without opening a browser.
    """
    print("\U0001F916 Local Voice Assistant")
    print("\u2500" * 42)
    selected_language = input("Which language should I use for transcription? (e.g. de, en, fr): ").strip().lower() or "en"

    print(f"TTS voice: {_tts_voice}")
    print(f"LLM model: {args.model or ('gemma3' if args.provider == 'ollama' else 'MiniMax-M2.7')}")
    print(f"LLM provider: {args.provider}")
    print("\u2500" * 42)
    print("Press Ctrl+C to exit.\n")

    if args.save_voice:
        os.makedirs("voices", exist_ok=True)

    response_count = 0
    _start_new_session()

    def speak(text: str):
        """
        Synthesize and play text out loud, sentence by sentence — the
        first sentence starts playing as soon as it's ready, while later
        sentences are still being synthesized, instead of waiting for the
        whole reply (see the project plan, Step 5 "reduce latency").

        Honors --save-voice by collecting the sentences actually played
        and writing them to disk afterwards as one file, instead of
        synthesizing everything a second time just for saving.
        """
        nonlocal response_count
        sample_rate = None
        played_chunks = []

        for sr, audio_array in get_tts().synthesize_streaming(text):
            sample_rate = sr
            speech.play_audio(sr, audio_array)
            if args.save_voice:
                played_chunks.append(audio_array)

        if args.save_voice and played_chunks:
            response_count += 1
            filename = f"voices/response_{response_count:03d}.wav"
            full_audio = np.concatenate(played_chunks)
            speech.write_wav_file(filename, sample_rate, full_audio)
            print(f"Voice saved to: {filename}")

    try:
        # The framing message and the first fixed question open the
        # interview — presented verbatim, no LLM call needed.
        first_question = interview.current_question()
        if first_question is None:
            print(f"Assistant: {interview.framing_message}")
            speak(interview.framing_message)
            print("(This interview guide has no questions yet.)")
            return
        opening = f"{interview.framing_message} {interview.presentation_text(first_question)}".strip()
        print(f"Assistant: {opening}")
        speak(opening)

        while not interview.is_finished():
            input("\U0001F3A4 Press Enter to start recording your answer, then press Enter again to stop.")

            data_queue: Queue = Queue()
            stop_event = threading.Event()
            recording_thread = threading.Thread(
                target=speech.record_audio,
                args=(stop_event, data_queue),
            )
            recording_thread.start()

            input()
            stop_event.set()
            recording_thread.join()

            audio_data = b"".join(list(data_queue.queue))
            audio_np = np.frombuffer(audio_data, dtype=np.int16).astype(np.float32) / 32768.0

            if audio_np.size == 0:
                print("No audio recorded. Please ensure your microphone is working.")
                continue

            print("Transcribing...")
            answer_text = get_stt().transcribe(audio_np, language=selected_language)
            print(f"You: {answer_text}")

            print("Generating response...")
            pieces = process_answer(answer_text)
            combined = " ".join(p["text"] for p in pieces).strip()

            print(f"Assistant: {combined}")
            speak(combined)

    except KeyboardInterrupt:
        print("\nExiting...")

    print("Session ended. Thank you for using the assistant!")


if __name__ == "__main__":
    if args.web:
        print("\U0001F916 Starting Flask web server for Local Voice Assistant")
        print(f"Open http://127.0.0.1:{args.port} in your browser to connect.")
        # threaded=True so the streamed /api/chat/audio response (Step 5b)
        # is served promptly rather than potentially waiting behind other
        # requests (e.g. the browser's static-file polling).
        app.run(host="127.0.0.1", port=args.port, threaded=True)
    else:
        run_terminal_mode()

def write_guide(test, yaml_text):
    """Write a temporary guide into the interviews folder and return its file name."""
    fd, path = tempfile.mkstemp(suffix=".yaml", dir=INTERVIEWS_DIR)
    with os.fdopen(fd, "w", encoding="utf-8") as f:
        f.write(yaml_text)
    test.addCleanup(os.remove, path)
    return os.path.basename(path)

class TestJudgmentAgentCalls(unittest.TestCase):

        def fake(prompt):
            """Scripted fake model used in this test."""
            calls.append(prompt.to_string())
            reply = replies[min(len(calls), len(replies)) - 1]
            if isinstance(reply, Exception):
                raise reply
            return reply
        result = judge_answer(RunnableLambda(fake), item, [{"prompt": item["text"], "answer": "Gut."}])
        return result, calls

                                          "missing_information": [], "reasoning": "ok"}) + "\n```"
        result, calls = self.run_with([reply])
        self.assertEqual(result.intent, "answer")
        self.assertEqual(len(calls), 1)

    def test_retries_once_then_succeeds(self):
        """Retries once then succeeds."""
        good = json.dumps({"intent": "answer", "sufficiency": "sufficient", "reasoning": "ok"})
        result, calls = self.run_with(["not json at all", good])
        self.assertEqual(len(calls), 2)
        self.assertEqual(result.sufficiency, "sufficient")

    def test_disallowed_values_rejected(self):

    "Ich verstehe die Frage nicht.": ("clarification_request", "insufficient", ["meaning of the question"], "", None),

    "Hotdogs machen mich nicht glücklich.": ("answer", "ambiguous", ["whether pleasure is reduced in general, not only for one food"], "Hotdogs", None),
    "Ich denke manchmal an Suizid.": ("answer", "sufficient", [], "Ich denke manchmal an Suizid", None),

QUESTION_AGENT_REPLIES = {
    "did not understand": "Gemeint ist, wie Sie sich gefühlsmäßig meistens gefühlt haben: Wie ging es Ihnen in den letzten zwei Wochen?",
    "missing detail": "Was meinen Sie genau mit diesem Gefühl?",
    "cannot answer": "Wenn Sie nur an die letzten Tage denken, wie war Ihre Stimmung da?",
    "unrelated": "Lassen Sie uns zur Frage zurückkehren: Wie haben Sie sich in den letzten zwei Wochen gefühlt?",
}


class ScriptedModel:
    """Fake model: routes each prompt to the scripted reply of its agent."""

    def __init__(self):
        """Start in normal mode and keep every prompt each agent received, so tests can inspect them."""
        self.mode = "ok"  # "ok" | "garbage" | "raise"
        self.judgment_prompts = []
        self.question_prompts = []
        self.ack_prompts = []

    def __call__(self, prompt):
        """Route the prompt to the scripted reply of the agent it belongs to (Judgment / Question / acknowledgement)."""
        text = prompt.to_string() if hasattr(prompt, "to_string") else str(prompt)
        if "You are the Judgment Agent" in text:
            self.judgment_prompts.append(text)
            if self.mode == "garbage":
                return "Sure! The participant seems fine."
            if self.mode == "raise":
                raise ConnectionError("model server not reachable")
            latest = re.findall(r"<<<(.*?)>>>", text, re.S)[-1]
            intent, suff, missing, quote, rating = SCRIPT[latest]
            return json.dumps({"intent": intent, "sufficiency": suff, "missing_information": missing,
                               "evidence_quote": quote, "provisional_rating": rating,
                               "reasoning": "scripted"})
        if "You are the Question Agent" in text:
            self.question_prompts.append(text)
            for key, reply in QUESTION_AGENT_REPLIES.items():
                if key in text:
                    return reply
            return "Können Sie das bitte genauer beschreiben?"
        self.ack_prompts.append(text)
        return "FEEDBACK: Danke für Ihre Antwort.\nFOLLOWUP: NONE"


class FlowTestBase(unittest.TestCase):
    """Starts a fresh interview on the 3-question development guide through the real Flask app, with a scripted model and a temporary record folder."""
    def setUp(self):
        """Prepare a fresh interview, scripted model and temporary record location for each test."""
        self.model = ScriptedModel()
        fake = RunnableLambda(self.model)
        app_module.llm = fake
        app_module.chain_with_history = llm_provider.build_chain(fake)
        self.records_dir = tempfile.mkdtemp()
        self.addCleanup(shutil.rmtree, self.records_dir, ignore_errors=True)
        # These flows use the 3-question development guide regardless of the
        # guide that is active in study_settings.yaml.
        app_module.interview = InterviewGuide("depression_scid_example.yaml")
        app_module.recorder = SessionRecorder(app_module.interview, directory=self.records_dir)
        self.client = app_module.app.test_client()
        opening = self.client.get("/api/interview/start").get_json()["message"]
        self.assertIn(Q1, opening)

    def say(self, text):
        """Send one participant message to /api/chat and return the JSON reply."""
        res = self.client.post("/api/chat", json={"text": text})
        self.assertEqual(res.status_code, 200)
        return res.get_json()

    def current(self):
        """Id of the item that is currently being asked (None at the end)."""
        q = app_module.interview.current_question()
        return q["id"] if q else None

    def records(self):
        """The item records collected so far."""
        return app_module.interview.records

    def saved_record(self):
        """Load the single session-record JSON file written during the test."""
        files = os.listdir(self.records_dir)
        self.assertEqual(len(files), 1)
        with open(os.path.join(self.records_dir, files[0]), encoding="utf-8") as f:
            return json.load(f)


class TestClearAnswers(FlowTestBase):
    """Clear positive, clear negative and short-but-sufficient answers."""
    def test_positive_negative_and_short_answers_complete_the_interview(self):
        """Positive negative and short answers complete the interview."""
        r = self.say("Sehr schlecht, ich bin fast jeden Tag niedergeschlagen.")
        self.assertIn(Q2, r["response"])
        r = self.say("Nein.")  # clear negative: no probing for a symptom
        self.assertIn(Q3, r["response"])
        r = self.say("Gut.")   # short but sufficient
        self.assertTrue(r["finished"])
        self.assertIn("letzte Frage", r["response"])
        self.assertEqual([x["status"] for x in self.records()], ["answered"] * 3)
        self.assertEqual([x["clarifications_used"] for x in self.records()], [0, 0, 0])
        self.assertEqual(self.model.question_prompts, [])
        saved = self.saved_record()
        self.assertTrue(saved["finished"])
        self.assertIn("NOT SCID", saved["content_source"])
        self.assertIn("Not a diagnosis", saved["notice"])

    def test_clear_negative_sleep_answer_is_accepted(self):
        """Clear negative sleep answer is accepted."""
        self.say("Gut.")
        self.say("Nein.")
        r = self.say("Nein, ich schlafe gut, keine Probleme.")
        self.assertTrue(r["finished"])
        self.assertEqual(self.records()[-1]["status"], "answered")
        self.assertEqual(self.records()[-1]["judgment"]["sufficiency"], "sufficient")


class TestClarifications(FlowTestBase):
    """Ambiguous answers, explanation requests, inability to answer, off-topic replies, narrow examples and budget exhaustion."""
    def test_ambiguous_answer_gets_one_focused_clarification(self):
        """Ambiguous answer gets one focused clarification."""
        r = self.say("Manchmal komisch.")
        self.assertEqual(self.current(), "dep_01")
        self.assertEqual(r["response"], QUESTION_AGENT_REPLIES["missing detail"])
        self.assertIn("what 'komisch' means", self.model.question_prompts[0])
        r = self.say("Ich meine eher müde und gereizt.")
        self.assertIn(Q2, r["response"])
        # The clarification answer was judged together with the first answer.
        self.assertIn("<<<Manchmal komisch.>>>", self.model.judgment_prompts[-1])
        rec = self.records()[0]
        self.assertEqual((rec["status"], rec["clarifications_used"], len(rec["turns"])), ("answered", 1, 2))

    # EMSAG: RT01: Repeated ambiguous answers do not trigger a numeric-limit skip.
    def test_repeated_ambiguous_answers_keep_current_item_active(self):
        """Repeated ambiguous answers keep the current item active."""
        self.say("Manchmal komisch.")
        r = self.say("Manchmal komisch.")

        self.assertEqual(self.current(), "dep_01")
        self.assertEqual(r["response"], QUESTION_AGENT_REPLIES["missing detail"])
        self.assertEqual(self.records(), [])
        self.assertEqual(app_module.interview.current_clarification_count(), 2)

    # EMSAG: RT01: Repeated explanation requests remain on the same item.
    def test_repeated_explanation_requests_keep_rephrasing(self):
        """Repeated explanation requests do not force the interview to skip."""
        for _ in range(3):
            r = self.say("Ich verstehe die Frage nicht.")
            self.assertEqual(self.current(), "dep_01")
            self.assertNotIn(Q1, r["response"])
            self.assertEqual(
                r["response"],
                QUESTION_AGENT_REPLIES["did not understand"],
            )

        self.assertEqual(self.records(), [])
        self.assertEqual(app_module.interview.current_clarification_count(), 3)

    # EMSAG: RT01: Repeated cannot-answer responses allow further gentle follow-ups.
    def test_repeated_cannot_answer_responses_keep_current_item_active(self):
        """Cannot-answer responses allow repeated gentle follow-ups."""
        for _ in range(3):
            r = self.say("Weiß ich nicht.")
            self.assertEqual(
                r["response"],
                QUESTION_AGENT_REPLIES["cannot answer"],
            )
            self.assertEqual(self.current(), "dep_01")

        self.assertEqual(self.records(), [])
        self.assertEqual(app_module.interview.current_clarification_count(), 3)

    # EMSAG: RT01: Repeated off-topic and unclear responses allow further redirects.
    def test_repeated_off_topic_responses_keep_current_item_active(self):
        """Repeated off-topic responses are redirected without a numeric limit."""
        for _ in range(3):
            r = self.say("Ich habe Hotdogs gegessen.")
            self.assertEqual(
                r["response"],
                QUESTION_AGENT_REPLIES["unrelated"],
            )
            self.assertEqual(self.current(), "dep_01")

        self.assertEqual(self.records(), [])
        self.assertEqual(app_module.interview.current_clarification_count(), 3)

    def test_narrow_example_is_not_treated_as_general_loss_of_pleasure(self):
        """Narrow example is not treated as general loss of pleasure."""
        self.say("Gut.")
        self.say("Hotdogs machen mich nicht glücklich.")
        self.assertEqual(self.current(), "dep_02")  # not advanced on a narrow example
        self.assertIn("not only for one food", self.model.question_prompts[-1])

class TestModelFailures(FlowTestBase):
    """Malformed or failed model responses must never advance the interview."""
    def test_malformed_output_does_not_advance_or_use_budget_and_pauses_after_three(self):
        """Malformed output does not advance or use budget and pauses after three."""
        self.model.mode = "garbage"
        for _ in range(2):
            r = self.say("Gut.")
            self.assertIn("technisches Problem", r["response"])
            self.assertIn(Q1, r["response"])
            self.assertEqual(self.current(), "dep_01")
            self.assertEqual(app_module.interview.current_clarification_count(), 0)
        r = self.say("Gut.")
        self.assertIn("pausiert", r["response"])
        self.assertEqual(self.records(), [])  # nothing recorded as a participant result
        events = self.saved_record()["system_events"]
        self.assertEqual(len(events), 3)
        self.assertTrue(all(e["kind"] == "judgment_failure" for e in events))
        # Recovery: model back, interview still paused until restarted.
        self.model.mode = "ok"
        self.assertIn("pausiert", self.say("Gut.")["response"])

    def test_temporary_failure_then_success_continues_normally(self):
        """Temporary failure then success continues normally."""
        self.model.mode = "raise"
        self.say("Gut.")
        self.model.mode = "ok"
        r = self.say("Gut.")
        self.assertIn(Q2, r["response"])
        rec = self.records()[0]
        self.assertEqual(len(rec["turns"]), 1)  # the failed attempt is not stored as evidence


class TestProvisionalRatings(unittest.TestCase):
    """Rating path with SYNTHETIC test items in SCID-5-PD format (not SCID wording)."""

    GUIDE = """
guide_id: synthetic_pd
language: de
rating_scheme: scid5pd
sections:
  - questions:
      - {id: T1, text: "Synthetische Testfrage eins?", criterion: "synthetic avoidance", rating_anchor: "2 = several examples"}
      - {id: T2, text: "Synthetische Testfrage zwei?", criterion: "synthetic"}
      - {id: T3, text: "Synthetische Testfrage drei?", criterion: "synthetic"}
"""

    def setUp(self):
        """Prepare a fresh interview, scripted model and temporary record location for each test."""
        fd, path = tempfile.mkstemp(suffix=".yaml", dir=INTERVIEWS_DIR)
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(self.GUIDE)
        self.addCleanup(os.remove, path)
        self.guide = InterviewGuide(os.path.basename(path))
        self.recorder = SessionRecorder(self.guide, enabled=False)
        self.llm = RunnableLambda(ScriptedModel())

    def test_ratings_are_provisional_and_skips_are_unknown(self):
        """Ratings are provisional and skips are unknown."""
        handle_turn(self.llm, self.guide, self.recorder,
                    "Ja, ich meide große Gruppen fast immer, zum Beispiel bei der Arbeit und im Verein.")
        handle_turn(self.llm, self.guide, self.recorder, "Ja, ich bin sehr selbstbewusst.")
        handle_turn(self.llm, self.guide, self.recorder, "überspringen")
        ratings = [r["provisional_rating"] for r in self.guide.records]
        self.assertEqual(ratings[0]["value"], "2")
        self.assertIn("provisional", ratings[0]["label"])
        # invented evidence ("I am never anxious") -> rating forced to "?"
        self.assertEqual(ratings[1]["value"], "?")
        self.assertEqual(ratings[2]["value"], "?")  # skipped is never "0"
        self.assertNotIn("total", json.dumps(self.recorder.to_dict()).lower())

from tests.test_interview_flow import FlowTestBase


class TestLiveRT01ClarificationBudget(FlowTestBase):
    """
    RT01: Clarification requests must not consume the budget.
    The first off-topic answer should trigger a redirection.
    The second consecutive off-topic answer may exhaust the budget
    and advance the interview.
    """

    GENERIC_NEXT = "Danke. Wir gehen zur nächsten Frage weiter."

    def test_clarifications_do_not_consume_budget_and_off_topic_is_limited(self):
        # ---------------------------------------------------------
        # Step 1: First clarification request
        # ---------------------------------------------------------
        response1 = self.say("Ich verstehe die Frage nicht.")
        item1 = self.current()

        assert response1.get("response", "").strip()
        assert self.current() == item1
        assert self.GENERIC_NEXT not in response1["response"]

        # ---------------------------------------------------------
        # Step 2: Second clarification request
        # Clarification/rephrasing must not consume the budget.
        # ---------------------------------------------------------
        response2 = self.say("Ich verstehe die Frage immer noch nicht.")
        
        assert response2.get("response", "").strip()
        assert self.current() == item1
        assert self.GENERIC_NEXT not in response2["response"]

        # ---------------------------------------------------------
        # Step 3: First off-topic answer
        # The chatbot should redirect the participant to the topic.
        # It must not skip the current item yet.
        # ---------------------------------------------------------
        response3 = self.say("Ich habe Hotdogs gegessen.")

        assert response3.get("response", "").strip()
        assert self.current() == item1, (
            "The first off-topic answer must not skip the current item."
        )
        assert self.GENERIC_NEXT not in response3["response"], (
            "The chatbot must not immediately say 'Danke, nächste Frage' "
            "after the first off-topic answer."
        )

        # ---------------------------------------------------------
        # Step 4: Second consecutive off-topic answer
        # The off-topic allowance is now exhausted; the interview
        # may move to the next item.
        # ---------------------------------------------------------
        response4 = self.say("Heute habe ich Pizza gegessen.")

        assert response4.get("response", "").strip()
        assert self.current() != item1, (
            "After the second consecutive off-topic answer, "
            "the current item should be skipped."
        )

def is_generic_thanks_only(text):
    """Detect an acknowledgement consisting only of generic thanks."""
    normalized = " ".join(
        (text or "").casefold().split()
    ).strip(" .!,:;")

    generic_responses = {
        "danke",
        "danke für ihre antwort",
        "vielen dank",
        "vielen dank für ihre antwort",
    }

    return normalized in generic_responses


def is_mechanical_transition(text):
    """Detect known generic-thanks transitions."""
    normalized = " ".join((text or "").casefold().split())

    mechanical_phrases = (
        "danke. wir gehen zur nächsten frage weiter",
        "danke für ihre antwort. wir gehen zur nächsten frage weiter",
        "vielen dank. wir gehen zur nächsten frage weiter",
        "vielen dank für ihre antwort. wir gehen zur nächsten frage weiter",
    )

    return any(
        phrase in normalized
        for phrase in mechanical_phrases
    )


def assert_empathetic_response(test_case, visible_text, scenario):
    """Shared quality checks for emotional acknowledgement."""
    normalized = " ".join((visible_text or "").casefold().split())

    test_case.assertTrue(
        visible_text and visible_text.strip(),
        f"{scenario}: the visible response must not be empty.",
    )

    test_case.assertFalse(
        is_mechanical_transition(normalized),
        f"{scenario}: a mechanical transition was shown: "
